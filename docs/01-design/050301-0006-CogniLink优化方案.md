# CogniLink插件技术架构优化方案

## 问题分析

### 1. 当前技术瓶颈
- **RAG存储性能**：JSON文件存储向量，检索效率O(n)，无法支持大规模数据
- **架构限制**：纯Obsidian插件形态限制AI/NLP能力发挥
- **内容质量**：缺乏针对社交媒体的专门优化和质量保障

### 2. 产品形态问题
- JavaScript/TypeScript环境难以充分利用Python AI生态
- Electron环境资源限制影响复杂AI计算
- 缺乏平台特定的内容适配机制

## 优化方案

### 一、产品形态重构：混合架构

**推荐架构：Obsidian插件UI + 本地Python服务器**

```
┌─────────────────┐    HTTP/WebSocket    ┌──────────────────┐
│  Obsidian插件   │ ←─────────────────→ │  Python AI服务器  │
│  (UI交互层)     │                     │  (AI处理层)      │
│  - 用户界面     │                     │  - RAG检索       │
│  - 笔记访问     │                     │  - 内容生成      │
│  - 结果展示     │                     │  - 质量评估      │
└─────────────────┘                     └──────────────────┘
```

```python
# 新的向量存储实现
import lancedb
import numpy as np
from sentence_transformers import SentenceTransformer

class OptimizedVectorStore:
    def __init__(self, db_path="./cognilink_vectors"):
        self.db = lancedb.connect(db_path)
        self.model = SentenceTransformer('all-MiniLM-L6-v2')
        self.table_name = "note_embeddings"
    
    def add_documents(self, documents):
        """批量添加文档向量"""
        embeddings = self.model.encode([doc.content for doc in documents])
        
        data = []
        for doc, embedding in zip(documents, embeddings):
            data.append({
                "id": doc.path,
                "content": doc.content,
                "title": doc.title,
                "tags": doc.tags or [],
                "updated": doc.updated,
                "vector": embedding.tolist(),
                "metadata": doc.metadata or {}
            })
        
        if self.table_exists():
            table = self.db.open_table(self.table_name)
            table.add(data)
        else:
            table = self.db.create_table(self.table_name, data)
        
        return len(data)
    
    def search(self, query, top_k=10, threshold=0.7):
        """语义搜索"""
        if not self.table_exists():
            return []
        
        query_vector = self.model.encode([query])[0]
        table = self.db.open_table(self.table_name)
        
        results = (table
                  .search(query_vector)
                  .limit(top_k)
                  .to_list())
        
        # 过滤低相似度结果
        filtered_results = [
            r for r in results 
            if r.get('_distance', 1) < (1 - threshold)
        ]
        
        return filtered_results
    
    def table_exists(self):
        return self.table_name in self.db.table_names()
```

**性能提升**：
- 从O(n)线性搜索提升到O(log n)索引搜索
- 支持增量更新，无需重建整个索引
- 内存使用优化，支持大规模数据

#### 2. 智能分块策略

```python
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.text_splitter import MarkdownHeaderTextSplitter

class SemanticChunker:
    def __init__(self):
        # Markdown结构感知分割
        self.header_splitter = MarkdownHeaderTextSplitter(
            headers_to_split_on=[
                ("#", "Header 1"),
                ("##", "Header 2"), 
                ("###", "Header 3"),
            ]
        )
        
        # 语义感知分割
        self.text_splitter = RecursiveCharacterTextSplitter(
            chunk_size=512,
            chunk_overlap=50,
            separators=["\n\n", "\n", ". ", " ", ""],
            length_function=len
        )
    
    def chunk_document(self, markdown_content, metadata=None):
        """智能分块，保持语义完整性"""
        chunks = []
        
        # 1. 按标题结构分割
        header_chunks = self.header_splitter.split_text(markdown_content)
        
        # 2. 进一步语义分割
        for chunk in header_chunks:
            content = chunk.page_content
            chunk_metadata = {**(metadata or {}), **chunk.metadata}
            
            if len(content) <= 512:
                chunks.append({
                    "content": content,
                    "metadata": chunk_metadata
                })
            else:
                # 大块进一步分割
                sub_chunks = self.text_splitter.split_text(content)
                for sub_chunk in sub_chunks:
                    chunks.append({
                        "content": sub_chunk,
                        "metadata": chunk_metadata
                    })
        
        return chunks
```

#### 3. 多阶段RAG检索

```python
class AdvancedRAGRetriever:
    def __init__(self, vector_store, keyword_weight=0.3):
        self.vector_store = vector_store
        self.keyword_weight = keyword_weight
    
    def retrieve(self, query, top_k=8, min_score=0.7):
        """多阶段检索：语义 + 关键词 + 重排序"""
        
        # 阶段1：语义检索
        semantic_results = self.vector_store.search(
            query, top_k=top_k*2, threshold=min_score
        )
        
        # 阶段2：关键词增强
        enhanced_results = self._enhance_with_keywords(
            semantic_results, query
        )
        
        # 阶段3：多样性重排序
        final_results = self._diversify_results(
            enhanced_results, top_k
        )
        
        return final_results
    
    def _enhance_with_keywords(self, results, query):
        """关键词匹配增强"""
        query_terms = set(query.lower().split())
        
        for result in results:
            content = result['content'].lower()
            title = result.get('title', '').lower()
            tags = ' '.join(result.get('tags', [])).lower()
            
            # 计算关键词匹配分数
            keyword_score = 0
            for term in query_terms:
                if term in content:
                    keyword_score += 1
                if term in title:
                    keyword_score += 2  # 标题权重更高
                if term in tags:
                    keyword_score += 1.5
            
            # 综合评分
            semantic_score = 1 - result.get('_distance', 1)
            combined_score = (
                semantic_score * (1 - self.keyword_weight) + 
                keyword_score * self.keyword_weight
            )
            result['combined_score'] = combined_score
        
        return sorted(results, key=lambda x: x['combined_score'], reverse=True)
    
    def _diversify_results(self, results, top_k):
        """结果多样性优化，避免重复内容"""
        selected = []
        similarity_threshold = 0.85
        
        for result in results:
            if len(selected) >= top_k:
                break
                
            # 检查与已选择内容的相似度
            is_diverse = True
            for selected_result in selected:
                similarity = self._calculate_similarity(
                    result['content'], 
                    selected_result['content']
                )
                if similarity > similarity_threshold:
                    is_diverse = False
                    break
            
            if is_diverse:
                selected.append(result)
        
        return selected
```
